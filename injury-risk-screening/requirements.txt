# ==============================================================================
# Real-Time Markerless AI for Personalized Sports Injury-Risk Screening
# Core dependencies for the hackathon MVP
#
# All versions below were checked for a prebuilt wheel on Python 3.13 on
# BOTH Windows and Linux before being pinned (not assumed) -- several of
# the original pins predated Python 3.13's release and had no wheel at
# all for it, which is what was actually causing `pip install -r
# requirements.txt` to hang or fail. See inline notes on each fix.
# ==============================================================================

# --- Video / Computer Vision ---
# opencv-python 5.x requires numpy>=2, which conflicted with numpy==1.26.4
# below -- pip's resolver would backtrack for minutes or fail outright.
# Not a Windows-specific issue; it reproduces identically on Linux.
opencv-python==4.10.0.84

# --- Pose Estimation ---
# Note: recent mediapipe releases dropped the legacy `mediapipe.solutions`
# API entirely -- this project uses the Tasks API (mediapipe.tasks.python.vision)
# throughout, which works on this version and is the actively supported path.
mediapipe==1.0.1

# --- Numerical / Data ---
# numpy==1.26.4 (the original pin) has NO wheel at all for Python 3.13 on
# Windows or Linux -- it predates 3.13's release. This is almost certainly
# the actual root cause of installs failing: pip either errors immediately
# ("No matching distribution found") or, combined with the opencv-python
# conflict above, spends minutes backtracking before failing. 2.1.3 has
# real cp313 wheels on both platforms and is what pandas 2.2.3/scipy 1.14.1
# below were tested against.
numpy==2.1.3
pandas==2.2.3                # 2.2.2 (original pin) has no cp313 wheel
scipy==1.14.1                # 1.13.1 (original pin) has no cp313 wheel

# --- ML: event-level tabular baseline model ---
xgboost==2.1.1
scikit-learn==1.5.2          # 1.5.1 (original pin) has no cp313 wheel
joblib==1.4.2                # model save/load (src/models/train_risk_model.py)

# --- Biomechanical reference/validation (src/validation/) ---
# Real OpenSim Python bindings -- used to build a genuine 6-DOF knee
# model and run real forward kinematics for the validation pipeline.
# Confirmed real cp313 wheels for both win_amd64 and manylinux_2_28.
opensim==4.6

# --- Backend / API ---
fastapi==0.115.0
uvicorn[standard]==0.30.6
pydantic==2.9.2

# --- Prototype UI ---
streamlit==1.38.0

# --- Database ---
psycopg2-binary==2.9.10      # PostgreSQL driver; 2.9.9 has no cp313 wheel
sqlalchemy==2.0.35
# SQLite is used via Python's built-in sqlite3 module for local prototyping

# --- Required for DataFrame.to_parquet() in scripts/run_pose_on_video.py ---
pyarrow==19.0.1               # 25.0.1 (original pin) does not exist on PyPI

# --- Storage / utils ---
python-dotenv==1.0.1
pyyaml==6.0.3                 # 6.0.2 (original pin) does not exist on PyPI

# --- HTTP (dashboard -> API calls, scripts/process_session.py, FastAPI TestClient) ---
requests==2.32.3
httpx==0.27.2

# --- Testing ---
pytest==8.3.3
pytest-cov==5.0.0

# --- Notebooks (optional; not required to run anything above) ---
jupyter==1.1.1
matplotlib==3.10.1            # 3.9.2 (original pin) has no cp313 wheel
